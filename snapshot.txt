import Sidebar from './components/Sidebar'
import Page from './components/Page'

import Login from './pages/Login'
import Onboarding from './pages/Onboarding'
import Dashboard from './pages/Dashboard'
import Transactions from './pages/Transactions'
import Investment from './pages/Investment'

function Shell() {
  const { user, profile, login } = useProfile()

  // Not logged in
  if (!user) {
    return (
      <Routes>
        <Route path="/" element={<Landing />} />
        <Route path="/login" element={<Login onAuth={login} />} />
        <Route path="*" element={<Navigate to="/" replace />} />
      </Routes>
    )
  }

  // Logged in, first-time onboarding
  if (!profile) return <Onboarding />

  // Logged in with profile
  return (
    <Routes>
      <Route path="/edit-profile" element={<Onboarding />} />
      <Route
        path="*"
        element={
          <div className="flex min-h-screen">
            <Sidebar />
            <main className="flex-1 overflow-y-auto p-8">
              <Routes>
                <Route path="/" element={<Dashboard />} />
                <Route path="/transactions" element={<Transactions />} />
                <Route path="/investments" element={<Investment />} />

              </Routes>

        }
      />
    </Routes>
  )
}

export const cagr = (returns: number[]): number => {
  const growth = returns.reduce((acc, r) => acc * (1 + r / 100), 1);
  return (Math.pow(growth, 1 / returns.length) - 1) * 100;
};

export const stats = (returns: number[]) => {
  const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
  const variance =
    returns.reduce((a, b) => a + (b - mean) ** 2, 0) / returns.length;
  const best = Math.max(...returns);
  const worst = Math.min(...returns);
  return {
    cagr: cagr(returns),
    best,
    bestFY: FY_LABELS[returns.indexOf(best)],
    worst,
    worstFY: FY_LABELS[returns.indexOf(worst)],
    volatility: Math.sqrt(variance),
    positiveYears: returns.filter((r) => r > 0).length,
  };
};

===== src\lib\finance.ts =====
import type { Profile, Goal } from '../types'

const sum = (n: number[]) => n.reduce((a, b) => a + b, 0)

export const short = (n: number) => {
  const a = Math.abs(n)
  const s = a >= 1e7 ? `₹${+(a / 1e7).toFixed(2)} Cr` : a >= 1e5 ? `₹${+(a / 1e5).toFixed(2)} L` : a >= 1e3 ? `₹${+(a / 1e3).toFixed(1)}K` : `₹${Math.round(a)}`
  return n < 0 ? `-${s}` : s
}

export const monthsUntil = (d: string) =>
  Math.max(1, Math.round((new Date(d).getTime() - Date.now()) / (1000 * 60 * 60 * 24 * 30.44)))

const yearlyReturn = (risk: Profile['risk']) => (risk === 'Conservative' ? 0.08 : risk === 'Aggressive' ? 0.13 : 0.11)

export function requiredSip(g: Goal, risk: Profile['risk']): number {
  if (!g.targetAmount || !g.targetDate) return 0
  const n = monthsUntil(g.targetDate)
  const r = yearlyReturn(risk) / 12
  return Math.round((g.targetAmount * r) / (Math.pow(1 + r, n) - 1))
}

export type VelocityStatus = 'Accelerating' | 'On Pace' | 'Lagging' | 'No goals'

export function getMetrics(p: Profile) {
  const income = p.income + (p.spouseIncome ?? 0) + (p.otherIncome ?? 0)
  const fixed = p.rentEmi + p.otherBills
  const spending = sum(Object.values(p.expenses ?? {})) + sum((p.customExpenses ?? []).map((i) => i.amount))
  const sip = p.monthlySip
  const budget = Math.max(0, income - fixed - sip)
  const left = budget - spending

  const goals = (p.goals ?? []).map((g) => ({ ...g, need: requiredSip(g, p.risk) }))
  const requiredTotal = sum(goals.map((g) => g.need))
  const ratio = requiredTotal > 0 ? sip / requiredTotal : 0
  const status: VelocityStatus =
    requiredTotal === 0 ? 'No goals' : ratio >= 1.1 ? 'Accelerating' : ratio >= 0.9 ? 'On Pace' : 'Lagging'

  const assetsTotal = sum(Object.values(p.assets ?? {})) + sum((p.customAssets ?? []).map((i) => i.amount))
  const netWorth = p.investments + assetsTotal - p.loans

  return { income, fixed, spending, sip, budget, left, goals, requiredTotal, ratio, status, assetsTotal, netWorth }
}

export const TX_KEY = 'kairos_transactions_v2'
const REM_KEY = 'kairos_reminder'

export type Reminder = {
  enabled: boolean
  time: string // "HH:MM" in 24h
  lastFired: string // date (YYYY-MM-DD) the reminder last showed
  lastNoted: string // date (YYYY-MM-DD) the user last noted transactions
}

const DEFAULT: Reminder = { enabled: false, time: '21:00', lastFired: '', lastNoted: '' }

const pad = (n: number) => String(n).padStart(2, '0')
export const isoOf = (d: Date) => `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}`
export const todayIso = () => isoOf(new Date())

export function loadReminder(): Reminder {
  try {
    const raw = localStorage.getItem(REM_KEY)
    return raw ? { ...DEFAULT, ...JSON.parse(raw) } : { ...DEFAULT }
  } catch {
    return { ...DEFAULT }
  }
}

export function updateReminder(patch: Partial<Reminder>): Reminder {
  const next = { ...loadReminder(), ...patch }
  localStorage.setItem(REM_KEY, JSON.stringify(next))
  window.dispatchEvent(new Event('kairos-reminder'))
  return next
}

export const markNoted = () => updateReminder({ lastNoted: todayIso() })

function daysToSalary(creditDay: number) {
  const now = new Date()
  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate())
  const credit = (y: number, m: number) => new Date(y, m, Math.min(creditDay, new Date(y, m + 1, 0).getDate()))
  let next = credit(now.getFullYear(), now.getMonth())
  if (next <= today) next = credit(now.getFullYear(), now.getMonth() + 1)
  return Math.max(1, Math.round((next.getTime() - today.getTime()) / 86400000))
}



function SafeToSpend({ left, creditDay }: { left: number; creditDay: number }) {
  const [mode, setMode] = useState<'total' | 'daily'>('daily')
  const days = daysToSalary(creditDay)
  const safe = Math.max(0, left)
  const value = mode === 'daily' ? safe / days : safe
  return (
    <div className="card p-6">

function project(monthly: number, lump: number, years: number, annualPct: number) {
  const mRate = Math.pow(1 + annualPct / 100, 1 / 12) - 1
  let balance = lump
  const out: number[] = [balance]
  for (let y = 1; y <= years; y++) {
    for (let m = 0; m < 12; m++) balance = (balance + monthly) * (1 + mRate)
    out.push(balance)
  }
  return out
}

import { useState } from 'react'
import { useSearchParams} from 'react-router-dom'
import { Mail, Lock, User, Eye, EyeOff } from 'lucide-react'
import Logo from '../components/Logo'

export default function Login({ onAuth }: { onAuth: (name: string, email: string) => void }) {
 const [params] = useSearchParams()
  const [mode, setMode] = useState<'login' | 'signup'>(params.get('mode') === 'signup' ? 'signup' : 'login')
  const [name, setName] = useState('')
  const [email, setEmail] = useState('')
  const [password, setPassword] = useState('')
  const [confirm, setConfirm] = useState('')
  const [show, setShow] = useState(false)
  const [error, setError] = useState('')
  

  const submit = (e: React.FormEvent) => {
    e.preventDefault()
    if (mode === 'signup' && !name.trim()) return setError('Enter your full name.')
    if (!/^\S+@\S+\.\S+$/.test(email)) return setError('Enter a valid email.')
    if (password.length < 6) return setError('Password must be at least 6 characters.')
    if (mode === 'signup' && password !== confirm) return setError('Passwords do not match.')
    setError('')
    onAuth(name.trim() || email.split('@')[0], email)
    
  }

  const field =
    'w-full bg-white/80 border border-slate-200 rounded-xl pl-10 pr-3 py-3 text-sm outline-none focus:border-brand-500 focus:ring-2 focus:ring-brand-500/20'

  return (
    <div className="min-h-screen flex">
      <div className="hidden lg:flex flex-1 bg-gradient-to-br from-ink-900 via-ink-800 to-brand-600 text-white p-14 flex-col justify-between">
        <Logo light size={56} />
        <div>
          <h1 className="text-4xl font-extrabold leading-tight">
            Your money,<br />on autopilot.
          </h1>
          <p className="mt-4 text-slate-300 max-w-md">
            Autonomous capital velocity steering. All AI runs on your device. Your data never leaves it.
          </p>
        </div>
        <div className="text-xs text-slate-400">100% offline AI · Zero data egress</div>
      </div>

      <div className="flex-1 flex items-center justify-center p-6">
        <form onSubmit={submit} className="card w-full max-w-md p-8 space-y-4">
          <h2 className="text-2xl font-bold">{mode === 'login' ? 'Welcome back' : 'Create your account'}</h2>
          <p className="text-sm text-slate-500">
            {mode === 'login' ? 'Log in to continue.' : 'Takes less than a minute.'}
          </p>

          {mode === 'signup' && (
            <div className="relative">
              <User size={16} className="absolute left-3 top-3.5 text-slate-400" />
              <input className={field} placeholder="Full name" value={name} onChange={(e) => setName(e.target.value)} />
            </div>
          )}

          <div className="relative">
            <Mail size={16} className="absolute left-3 top-3.5 text-slate-400" />
            <input className={field} placeholder="Email" value={email} onChange={(e) => setEmail(e.target.value)} />
          </div>

          <div className="relative">
            <Lock size={16} className="absolute left-3 top-3.5 text-slate-400" />
            <input
              className={field}
              type={show ? 'text' : 'password'}
              placeholder="Password"
              value={password}
              onChange={(e) => setPassword(e.target.value)}
            />
            <button type="button" onClick={() => setShow(!show)} className="absolute right-3 top-3.5 text-slate-400">
              {show ? <EyeOff size={16} /> : <Eye size={16} />}
            </button>
          </div>

          {mode === 'signup' && (
            <div className="relative">
              <Lock size={16} className="absolute left-3 top-3.5 text-slate-400" />
              <input
                className={field}
                type="password"
                placeholder="Confirm password"
                value={confirm}
                onChange={(e) => setConfirm(e.target.value)}
              />
            </div>
          )}

          {error && <p className="text-sm text-red-500">{error}</p>}

          <button className="w-full bg-brand-600 hover:bg-brand-500 text-white font-semibold py-3 rounded-xl transition">
            {mode === 'login' ? 'Log in' : 'Sign up'}
          </button>

              type="button"
              onClick={() => addNamed(s)}
              className="px-2.5 py-1 rounded-full text-xs bg-slate-100 hover:bg-brand-500 hover:text-white text-slate-600 transition"
            >
              + {s}
            </button>
          ))}
        </div>
      )}
      {items.map((i) => (
        <div key={i.id} className="flex gap-2 items-center">
          <input
            className={`${inputCls} flex-1`}
            placeholder="Name (e.g. Education)"
            value={i.name}
            onChange={(e) => update(i.id, { name: e.target.value })}
          />
          <div className="relative w-40">
            <span className="absolute left-3 top-2.5 text-slate-400 text-sm">₹</span>
            <input
              inputMode="numeric"
              placeholder="0"
              className={`${inputCls} pl-7`}
              value={i.amount ? inr(i.amount) : ''}
              onChange={(e) => update(i.id, { amount: Number(e.target.value.replace(/\D/g, '')) })}
            />
          </div>
          <button type="button" onClick={() => onChange(items.filter((x) => x.id !== i.id))} className="text-slate-400 hover:text-red-500">
            <Trash2 size={16} />
          </button>
        </div>
      ))}
      <button
        type="button"
        onClick={() => onChange([...items, { id: crypto.randomUUID(), name: '', amount: 0 }])}
        className="flex items-center gap-1.5 text-sm font-semibold text-brand-600"
      >
        <Plus size={16} /> {title}
      </button>
    </div>
  )
}

import { createContext, useContext, useEffect, useState, type ReactNode } from 'react'
import type { Profile } from '../types'

interface User {
  name: string
  email: string
}

interface ProfileCtx {
  user: User | null
  profile: Profile | null
  login: (name: string, email: string) => void
  saveProfile: (p: Profile) => void
  logout: () => void
  resetProfile: () => void
}

const Ctx = createContext<ProfileCtx | null>(null)
const USER_KEY = 'kairos_user'
const profileKey = (email: string) => `kairos_profile_${email.trim().toLowerCase()}`

function load<T>(key: string): T | null {
  try {
    const raw = localStorage.getItem(key)
    return raw ? (JSON.parse(raw) as T) : null
  } catch {
    return null
  }
}

export function ProfileProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(() => load<User>(USER_KEY))
  const [profile, setProfile] = useState<Profile | null>(() => {
    const u = load<User>(USER_KEY)
    return u ? load<Profile>(profileKey(u.email)) : null
  })

  useEffect(() => {
    if (user) localStorage.setItem(USER_KEY, JSON.stringify(user))
    else localStorage.removeItem(USER_KEY)
  }, [user])

  useEffect(() => {
    if (user && profile) localStorage.setItem(profileKey(user.email), JSON.stringify(profile))
  }, [user, profile])

  return (
    <Ctx.Provider
      value={{
        user,
        profile,
        login: (name, email) => {
          setUser({ name, email })
          setProfile(load<Profile>(profileKey(email)))
        },
        saveProfile: setProfile,
        resetProfile: () => {
          if (user) localStorage.removeItem(profileKey(user.email))
          setProfile(null)
        },
        logout: () => {
          setUser(null)
          setProfile(null)
        },
      }}
    >
      {children}
    </Ctx.Provider>
  )
}

export function useProfile() {
  const c = useContext(Ctx)
  if (!c) throw new Error('useProfile must be used inside ProfileProvider')
  return c
}
